1. Get the frontend LoadBalancer URL:

   {{- if eq .Values.frontend.service.type "LoadBalancer" }}
   export FRONTEND_SVC_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "todo-bot.fullname" . }}-frontend-service --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
   echo "Frontend: http://$FRONTEND_SVC_IP:{{ .Values.frontend.service.port }}"
   {{- end }}
{{- else if eq .Values.frontend.service.type "NodePort" }}
1. Get the frontend NodePort URL:
   export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "todo-bot.fullname" . }}-frontend-service)
   export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
   echo "Frontend: http://$NODE_IP:$NODE_PORT"
{{- else }}
1. Access the frontend service by port forwarding:
   kubectl port-forward --namespace {{ .Release.Namespace }} svc/{{ include "todo-bot.fullname" . }}-frontend-service {{ .Values.frontend.service.port }}:{{ .Values.frontend.service.targetPort }}
   Then visit http://localhost:{{ .Values.frontend.service.port }}
